# Podman Quadlet — database pod (Postgres + MongoDB).
#
# Location on the server (rootless):
#   ~/.config/containers/systemd/myapp-db-pod.pod
#
# Generated systemd unit:
#   myapp-db-pod-pod.service
#
# Sits ONLY on myapp-data-net (internal network). Caddy and the public
# internet cannot reach it; only the app pod (also on myapp-data-net) can.
# Holds two containers sharing this pod's network namespace —
# myapp-postgres (5432) and myapp-mongo (27017). Because they're inside
# the same pod, they're addressable from each other as `localhost:<port>`
# AND from the app pod via their NetworkAlias names (below).

[Unit]
Description=myapp database pod (internal only)
Requires=myapp-data-net-network.service
After=myapp-data-net-network.service

[Pod]
PodName=myapp-db-pod
Network=myapp-data-net

# Keep the pod alive across container restarts. Default `stop` would
# kill the pod whenever the DB happens to be stopped, stranding it for
# the next start.
#
# `PodmanArgs=--exit-policy=continue` rather than `ExitPolicy=continue`
# because some Quadlet versions don't recognize the dedicated directive
# in [Pod]. PodmanArgs= forwards the literal flag to `podman pod create`.
PodmanArgs=--exit-policy=continue

# Aardvark-dns registers pods, not the containers inside them, so by
# default only `myapp-db-pod` resolves on the network. Add per-DB
# aliases so connection strings can name the actual database service
# (`myapp-postgres:5432`, `myapp-mongo:27017`) instead of the pod.
NetworkAlias=myapp-postgres
NetworkAlias=myapp-mongo

[Service]
# Auto-recover if a container restart drags the pod into a stopped
# state via Quadlet's BindsTo=.
Restart=on-failure
RestartSec=5s

[Install]
WantedBy=default.target
