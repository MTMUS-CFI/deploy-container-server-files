# Caddy site block for myapp.
#
# Caddy is the shared edge proxy on the server (set up once with the first
# app). For each new app, ADD a block to the existing
# `~/deploy/caddy/Caddyfile` — do NOT replace it.
#
# Two phases:
#   1. Before DNS — users reach the app at http://<server-ip>:<site-port>,
#      a host port unique to this app (the "site port"). See the README
#      "Ports" section to pick one, fill it in (./set-port.sh), publish it
#      from Caddy, and open the firewall.
#   2. After DNS — replace the port block with a hostname block so Caddy
#      auto-provisions a TLS cert.
#
# After editing the Caddyfile on the server:
#   podman exec caddy caddy validate --config /etc/caddy/Caddyfile
#   podman exec caddy caddy reload   --config /etc/caddy/Caddyfile
#
# `validate` first — `reload` will refuse a broken config and the running
# Caddy keeps serving the previous one.

# -----------------------------------------------------------------------------
# Phase 1 — site port (what users type in place of a domain name)
# -----------------------------------------------------------------------------
# The port is filled in by ./set-port.sh. If the block below still has the
# site-port placeholder, `caddy validate` rejects it, so it can't slip
# through unnoticed.
#
# A site address with only a port makes Caddy serve plain HTTP on that port
# for any address, so http://<server-ip>:<site-port>/ works.
#
# `myapp:3000` is the app's INTERNAL port, not the site port — leave it
# unless you changed the app port (README "Ports").

:<site-port> {
    encode zstd gzip

    reverse_proxy myapp:3000 {
        header_up Host {host}
        header_up X-Real-IP {remote_host}
        header_up X-Forwarded-For {remote_host}
        header_up X-Forwarded-Proto {scheme}

        transport http {
            read_timeout 5m
            write_timeout 5m
            dial_timeout 10s
        }
    }

    handle /healthz {
        respond "ok" 200
    }
}
